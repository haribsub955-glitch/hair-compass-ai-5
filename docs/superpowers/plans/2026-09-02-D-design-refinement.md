**Goal:** Keep the palette and typefaces but calm the hierarchy — semibold serif, a display scale capped at 40, quieter eyebrows, buttons and shadows, an 8-point rhythm per tab — and add seven new gouache illustrations in the existing style for the surfaces that have none.

//
//  TypeScaleTests.swift
//  Hair Compass AI 5Tests
//
//  The calmer hierarchy is a rule, not a moment: no serif headline in the app is set above
//  40 points, and the two weight tokens stay where the refinement put them. Scans the app
//  sources so a later edit cannot re-bolden the scale silently.
//

import Foundation
import Testing
@testable import Hair_Compass_AI_5

struct TypeScaleTests {

    /// The app source root, located from this file's compile-time path.
    private var appRoot: URL {
        URL(fileURLWithPath: #filePath)
            .deletingLastPathComponent()          // Hair Compass AI 5Tests
            .deletingLastPathComponent()          // repo root
            .appendingPathComponent("Hair Compass AI 5")
    }

    private func swiftSources() throws -> [(path: String, text: String)] {
        let fm = FileManager.default
        guard let e = fm.enumerator(at: appRoot, includingPropertiesForKeys: nil) else { return [] }
        var out: [(String, String)] = []
        for case let url as URL in e where url.pathExtension == "swift" {
            out.append((url.path, try String(contentsOf: url, encoding: .utf8)))
        }
        return out
    }

    @Test func noHeadlineAboveForty() throws {
        let pattern = try NSRegularExpression(pattern: #"Clinical\.headline\((\d+)"#)
        var offenders: [String] = []
        for (path, text) in try swiftSources() {
            let ns = text as NSString
            for m in pattern.matches(in: text, range: NSRange(location: 0, length: ns.length)) {
                let size = Int(ns.substring(with: m.range(at: 1))) ?? 0
                if size > 40 { offenders.append("\(path.split(separator: "/").last ?? ""): \(size)") }
            }
        }
        #expect(offenders.isEmpty, "headline sizes above 40: \(offenders)")
    }

    @Test func headlineDefaultsToSemibold() throws {
        let clinical = try String(contentsOf: appRoot.appendingPathComponent("Design/Clinical.swift"), encoding: .utf8)
        #expect(clinical.contains("static func headline(_ size: CGFloat, weight: Font.Weight = .semibold)"))
        #expect(clinical.contains("weight: .medium).monospaced()"), "eyebrow token must be medium")
    }
}

```python
#!/usr/bin/env python3
"""Generate the app's gouache illustrations with Gemini and cut them out for the asset catalog.

Reads the Gemini key from ~/.claude.json (mcpServers.nano-banana.env.GEMINI_API_KEY) at run
time. Never prints it. Never commit a key. Usage:

    python3 tools/generate_art.py --out "Hair Compass AI 5/Assets.xcassets" --sheet /path/contact.png [names...]

With no names, every piece in PIECES is generated.
"""
import argparse, base64, json, os, sys, time, urllib.request
from pathlib import Path
from PIL import Image, ImageDraw, ImageFilter

MODEL = "gemini-3-pro-image"
ENDPOINT = f"https://generativelanguage.googleapis.com/v1beta/models/{MODEL}:generateContent"
STYLE = ("soft gouache illustration, warm cream and copper palette with sage green accents, "
         "NO heavy outline — forms described by pigment and tone, not ink borders, "
         "subject isolated on a flat pure white background, no shadow on the ground, no text")

PIECES = {
    "art-starting-plan": ("a small brass compass resting on three folded paper cards, a sprig of green leaves beside them", None),
    "art-plan-checklist": ("THE SAME BIRD as the reference image, a small warm-brown wren, perched on the corner of a folded paper checklist card that shows three soft painted check marks", "wren-greeting"),
    "art-compare-locked": ("two single hair strands drawn side by side as gentle curves, one in full copper tone and one faint and unfinished, botanical", None),
    "art-learn": ("an open field guide book with a pressed green leaf between its pages", None),
    "art-in-clinic": ("a simple clinic chair beside a small potted plant on a low table", None),
    "art-procedures": ("a folded appointment card with a small painted calendar mark and a sprig of leaves", None),
    "wren-pointing": ("THE SAME BIRD as the reference image, a small warm-brown wren, full body, turned to the side with one wing raised and pointing toward the left", "wren-greeting"),
}

def key():
    cfg = json.load(open(os.path.expanduser("~/.claude.json")))
    return cfg["mcpServers"]["nano-banana"]["env"]["GEMINI_API_KEY"]

def generate(prompt, reference_png=None, aspect="3:2"):
    parts = [{"text": f"{prompt}, {STYLE}"}]
    if reference_png:
        parts.append({"inlineData": {"mimeType": "image/png",
                                     "data": base64.b64encode(Path(reference_png).read_bytes()).decode()}})
    body = {"contents": [{"parts": parts}],
            "generationConfig": {"responseModalities": ["IMAGE"], "imageConfig": {"aspectRatio": aspect}}}
    req = urllib.request.Request(ENDPOINT, data=json.dumps(body).encode(),
                                 headers={"x-goog-api-key": key(), "Content-Type": "application/json"})
    for attempt in range(3):
        try:
            with urllib.request.urlopen(req, timeout=180) as r:
                data = json.load(r)
            for part in data["candidates"][0]["content"]["parts"]:
                if "inlineData" in part:
                    return base64.b64decode(part["inlineData"]["data"])
            raise RuntimeError("no image part in response")
        except Exception as e:  # noqa: BLE001
            if attempt == 2:
                raise
            time.sleep(4)

def cut_out(png_bytes, tmp):
    """White background → alpha: flood fill from the border, then a tight pass, then feather."""
    tmp.write_bytes(png_bytes)
    im = Image.open(tmp).convert("RGBA")
    w, h = im.size
    # 1. Flood fill from every border pixel: paint reachable near-white to a marker colour.
    fill = im.copy()
    marker = (255, 0, 255, 255)
    for x in range(w):
        for y in (0, h - 1):
            if fill.getpixel((x, y))[:3] >= (235, 235, 235):
                ImageDraw.floodfill(fill, (x, y), marker, thresh=40)
    for y in range(h):
        for x in (0, w - 1):
            if fill.getpixel((x, y))[:3] >= (235, 235, 235):
                ImageDraw.floodfill(fill, (x, y), marker, thresh=40)
    px = im.load(); fp = fill.load()
    for y in range(h):
        for x in range(w):
            if fp[x, y] == marker:
                r, g, b, _ = px[x, y]
                px[x, y] = (r, g, b, 0)
    # 2. Tight pass: opaque pixels within 10 of pure white (enclosed background) go clear.
    for y in range(h):
        for x in range(w):
            r, g, b, a = px[x, y]
            if a and r >= 245 and g >= 245 and b >= 245:
                px[x, y] = (r, g, b, 0)
    # 3. Feather the alpha ~0.5 px.
    alpha = im.getchannel("A").filter(ImageFilter.GaussianBlur(0.5))
    im.putalpha(alpha)
    return im

def write_imageset(im, catalog, name):
    d = Path(catalog) / f"{name}.imageset"
    d.mkdir(parents=True, exist_ok=True)
    im.save(d / f"{name}.png")
    (d / "Contents.json").write_text(json.dumps({
        "images": [
            {"filename": f"{name}.png", "idiom": "universal", "scale": "1x"},
            {"idiom": "universal", "scale": "2x"},
            {"idiom": "universal", "scale": "3x"},
        ],
        "info": {"author": "xcode", "version": 1},
    }, indent=2) + "\n")

def contact_sheet(images, path):
    cell = 320
    sheet = Image.new("RGBA", (cell * len(images), cell * 2), (0, 0, 0, 0))
    for i, (name, im) in enumerate(images):
        thumb = im.copy(); thumb.thumbnail((cell - 20, cell - 20))
        for row, ground in enumerate(((251, 246, 239, 255), (43, 33, 26, 255))):
            tile = Image.new("RGBA", (cell, cell), ground)
            tile.alpha_composite(thumb, ((cell - thumb.width) // 2, (cell - thumb.height) // 2))
            sheet.alpha_composite(tile, (i * cell, row * cell))
    sheet.save(path)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", required=True)
    ap.add_argument("--sheet", required=True)
    ap.add_argument("names", nargs="*")
    a = ap.parse_args()
    names = a.names or list(PIECES)
    tmp = Path(a.sheet).with_suffix(".raw.png")
    done = []
    for name in names:
        prompt, ref = PIECES[name]
        ref_png = Path(a.out) / f"{ref}.imageset" / f"{ref}.png" if ref else None
        print(f"generating {name} …", flush=True)
        im = cut_out(generate(prompt, ref_png), tmp)
        write_imageset(im, a.out, name)
        done.append((name, im))
        print(f"  wrote {name}.imageset {im.size}")
    contact_sheet(done, a.sheet)
    print(f"contact sheet: {a.sheet}")

if __name__ == "__main__":
    sys.exit(main())
```
